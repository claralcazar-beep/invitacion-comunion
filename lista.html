-- Ejecutar COMPLETO en Supabase → SQL Editor → Run (proyecto comunion-juanes) -- Permite editar y cancelar confirmaciones desde el panel (botones Canceló y Editar). create or replace function public.plan_guardar(clave text, tabla text, fila jsonb) returns bigint language plpgsql security definer set search_path = public as $$ declare nuevo bigint; begin if not public.clave_ok(clave) then return null; end if; if tabla = 'gastos' then if fila ? 'id' and (fila->>'id') <> '' then update gastos set nombre=fila->>'nombre', categoria=coalesce(fila->>'categoria','Otros'), tipo=coalesce(fila->>'tipo','comun'), valor=coalesce((fila->>'valor')::numeric,0), proveedor=nullif(fila->>'proveedor',''), contacto=nullif(fila->>'contacto',''), encargado=nullif(fila->>'encargado',''), fecha_entrega=nullif(fila->>'fecha_entrega','')::date, estado=coalesce(fila->>'estado','pendiente'), notas=nullif(fila->>'notas','') where id=(fila->>'id')::bigint returning id into nuevo; else insert into gastos (nombre,categoria,tipo,valor,proveedor,contacto,encargado,fecha_entrega,estado,notas) values (fila->>'nombre', coalesce(fila->>'categoria','Otros'), coalesce(fila->>'tipo','comun'), coalesce((fila->>'valor')::numeric,0), nullif(fila->>'proveedor',''), nullif(fila->>'contacto',''), nullif(fila->>'encargado',''), nullif(fila->>'fecha_entrega','')::date, coalesce(fila->>'estado','pendiente'), nullif(fila->>'notas','')) returning id into nuevo; end if; elsif tabla = 'abonos' then insert into abonos (gasto_id, familia_id, fecha, valor, nota) values ((fila->>'gasto_id')::bigint, (fila->>'familia_id')::smallint, coalesce(nullif(fila->>'fecha','')::date, current_date), (fila->>'valor')::numeric, nullif(fila->>'nota','')) returning id into nuevo; elsif tabla = 'familias' then update familias set personas=coalesce((fila->>'personas')::int, personas), ninos=coalesce((fila->>'ninos')::smallint, ninos) where id=(fila->>'id')::smallint returning id into nuevo; elsif tabla = 'invitados' then update invitados set familia_id=coalesce((fila->>'familia_id')::smallint, familia_id), confirmado = case when fila ? 'confirmado' then (fila->>'confirmado')::boolean else confirmado end, personas = case when fila ? 'personas' then nullif(fila->>'personas','')::smallint else personas end, adultos = case when fila ? 'adultos' then nullif(fila->>'adultos','')::smallint else adultos end, ninos = case when fila ? 'ninos' then nullif(fila->>'ninos','')::smallint else ninos end, nombre = case when fila ? 'nombre' and trim(fila->>'nombre') <> '' then trim(fila->>'nombre') else nombre end, telefono = case when fila ? 'telefono' then nullif(trim(fila->>'telefono'),'') else telefono end, notas = case when fila ? 'notas' then nullif(trim(fila->>'notas'),'') else notas end where id=(fila->>'id')::bigint returning id into nuevo; elsif tabla = 'confirmaciones' then update confirmaciones set nombre = case when fila ? 'nombre' and trim(fila->>'nombre') <> '' then trim(fila->>'nombre') else nombre end, telefono = case when fila ? 'telefono' then nullif(trim(fila->>'telefono'),'') else telefono end, asiste = case when fila ? 'asiste' then (fila->>'asiste')::boolean else asiste end, adultos = case when fila ? 'adultos' then coalesce(nullif(fila->>'adultos','')::smallint,0) else adultos end, ninos = case when fila ? 'ninos' then coalesce(nullif(fila->>'ninos','')::smallint,0) else ninos end, mensaje = case when fila ? 'mensaje' then nullif(trim(fila->>'mensaje'),'') else mensaje end where id=(fila->>'id')::bigint returning id into nuevo; end if; return nuevo; end; $$;
